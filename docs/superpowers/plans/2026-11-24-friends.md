> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

Never check a partial step; never delete a finished step or a drift entry. Under subagent-driven development, `.superpowers/sdd/progress.md` is the runtime authority; the frontmatter mirrors it.

**Goal:** A member types the exact email of another account on `/friends` and sends a friend request. The other account sees it under « Demandes reçues » and accepts or refuses it; the sender can cancel it while it waits. Once accepted, the friendship is mutual and either side can remove it, after a confirmation. The friendships service exposes `areFriends(userId, otherUserId)`, which step 5 will receive to send photos only between friends. The admin is a member like any other here.

**Architecture:** A new `Friendship` collection, one document per pair (`requesterId`, `addresseeId`, `status` ∈ `pending | accepted`, `pairKey` = the two ids sorted and joined by `:`, `@unique`, so the index itself refuses a pair in either order). A new feature folder `server/src/features/friendships/` (repository, service with injected `repository` and `logger`, controller, routes) serves `GET /api/friendships`, `POST /api/friendships`, `POST /api/friendships/:id/accept` and `DELETE /api/friendships/:id`, all behind `requireAuth` only. Refuse, cancel and remove are the same delete, allowed to either side; the log names which one happened. `parseEmail` moves to `server/src/lib/email.ts`, shared with invitations. No real time: the page reloads its lists after each action. The client gets `friendship.ts`, `friendshipsApi.ts`, `useFriendships`, `useFriendshipActions` and `FriendsPage` with its components. Four PRs in a `gh stack`.

**Tech Stack:** Node 24, TypeScript 7 (`nodenext`, `strict`, `verbatimModuleSyntax`), Express 5, Prisma 6.19 (MongoDB), Vitest 5, Biome 2, Bruno CLI 4.2.0, React 19 + react-router 7 + shadcn/ui (`AlertDialog` already in `client/src/components/ui/alert-dialog.tsx`) + lucide-react (`UsersIcon`), `gh stack`.

- **Client messages (exact):** `USER_NOT_FOUND`: « Aucun compte n'utilise cette adresse e-mail. Demandez à l'administrateur d'inviter cette personne. »; `SELF_FRIEND`: « Vous ne pouvez pas vous ajouter vous-même. »; `ALREADY_FRIENDS`: « Vous êtes déjà amis. »; `FRIEND_REQUEST_EXISTS`: « Une demande est déjà en attente avec cette personne. - **DTOs (exact):** `FriendUser = { id, displayName, email }`, never `role` or the password hash (the repository selects only these three fields). `Friendship = { id, user: FriendUser, createdAt }`: `id` is the friendship's, `user` the other side, `createdAt` an ISO string. `FriendshipList = { friends, incoming, outgoing }`, each newest first (`id` descending), no pagination. Request answers `201 { friendship }`; list `200 { friends, incoming, outgoing }`; accept `200 { friendship }`; delete `204` without a body.
- **Rules (exact):** `pairKey` = `[userId, otherUserId].sort().join(':')`, computed by the repository only. Request: `parseEmail` → `EMAIL_INVALID`; no account → `USER_NOT_FOUND`; own id → `SELF_FRIEND`; existing row `accepted` → `ALREADY_FRIENDS`, `pending` (either direction) → `FRIEND_REQUEST_EXISTS`; `create` returning `null` (P2002, concurrent request) → `FRIEND_REQUEST_EXISTS`. Accept: one `update` filtered on `{ id, addresseeId, status: 'pending' }`, `null` (P2025) → `FRIENDSHIP_NOT_FOUND`. Delete: one `delete` filtered on `{ id, OR: [requester, addressee] }`, `null` → `FRIENDSHIP_NOT_FOUND`. An id that is not an ObjectId (`OBJECT_ID_PATTERN` = `^[0-9a-f]{24}$`, case-insensitive) → `FRIENDSHIP_NOT_FOUND` without a query. `areFriends` = the pair's row exists and is `accepted`.

import { Prisma, type PrismaClient } from '@prisma/client'

// The other side's public fields only: the password hash never leaves the repository.
const PARTY_FIELDS = { id: true, email: true, displayName: true } satisfies Prisma.UserSelect
const WITH_PARTIES = {
  requester: { select: PARTY_FIELDS },
  addressee: { select: PARTY_FIELDS },
} satisfies Prisma.FriendshipInclude

export type FriendshipRecord = Prisma.FriendshipGetPayload<{ include: typeof WITH_PARTIES }>

export type NewFriendship = { requesterId: string; addresseeId: string }

export type FriendshipsRepository = {
  findUserIdByEmail: (email: string) => Promise<string | null>
  findBetween: (userId: string, otherUserId: string) => Promise<FriendshipRecord | null>
  // null when the pair already has a row (a concurrent request): the service decides the error.
  create: (friendship: NewFriendship) => Promise<FriendshipRecord | null>
  findAllOf: (userId: string) => Promise<FriendshipRecord[]>
  // null unless id is a pending request addressed to addresseeId.
  acceptReceived: (id: string, addresseeId: string) => Promise<FriendshipRecord | null>
  // null unless userId is one of the two sides.
  deleteOf: (id: string, userId: string) => Promise<FriendshipRecord | null>
}

// P2002: a unique index refused the write. P2025: no record matched the filter.
type PrismaErrorCode = 'P2002' | 'P2025'

async function nullOnPrismaError<Result>(
  code: PrismaErrorCode,
  query: Promise<Result>,
): Promise<Result | null> {
  try {
    return await query
  } catch (error: unknown) {
    if (error instanceof Prisma.PrismaClientKnownRequestError && error.code === code) {
      return null
    }
    throw error
  }
}

// MongoDB cannot index "this pair in either order": sorted, the two ids make one key per pair.
function pairKeyOf(userId: string, otherUserId: string): string {
  return [userId, otherUserId].sort().join(':')
}

function eitherSide(userId: string): Prisma.FriendshipWhereInput[] {
  return [{ requesterId: userId }, { addresseeId: userId }]
}

function createPending(
  prisma: PrismaClient,
  { requesterId, addresseeId }: NewFriendship,
): Promise<FriendshipRecord | null> {
  const pairKey = pairKeyOf(requesterId, addresseeId)
  return nullOnPrismaError(
    'P2002',
    prisma.friendship.create({
      data: { requesterId, addresseeId, pairKey, status: 'pending' },
      include: WITH_PARTIES,
    }),
  )
}

// One atomic write: Prisma accepts the addressee and the status next to the unique id.
function acceptReceived(
  prisma: PrismaClient,
  id: string,
  addresseeId: string,
): Promise<FriendshipRecord | null> {
  return nullOnPrismaError(
    'P2025',
    prisma.friendship.update({
      where: { id, addresseeId, status: 'pending' },
      data: { status: 'accepted' },
      include: WITH_PARTIES,
    }),
  )
}

function deleteOf(
  prisma: PrismaClient,
  id: string,
  userId: string,
): Promise<FriendshipRecord | null> {
  return nullOnPrismaError(
    'P2025',
    prisma.friendship.delete({ where: { id, OR: eitherSide(userId) }, include: WITH_PARTIES }),
  )
}

export function createFriendshipsRepository(prisma: PrismaClient): FriendshipsRepository {
  return {
    findUserIdByEmail: async (email) => {
      const user = await prisma.user.findUnique({ where: { email }, select: { id: true } })
      return user?.id ?? null
    },
    findBetween: (userId, otherUserId) =>
      prisma.friendship.findUnique({
        where: { pairKey: pairKeyOf(userId, otherUserId) },
        include: WITH_PARTIES,
      }),
    create: (friendship) => createPending(prisma, friendship),
    // By id: an ObjectId starts with its creation time, so this is newest first.
    findAllOf: (userId) =>
      prisma.friendship.findMany({
        where: { OR: eitherSide(userId) },
        include: WITH_PARTIES,
        orderBy: { id: 'desc' },
      }),
    acceptReceived: (id, addresseeId) => acceptReceived(prisma, id, addresseeId),
    deleteOf: (id, userId) => deleteOf(prisma, id, userId),
  }
}

import { AppError } from '../../errors/app-error.js'
import { parseEmail } from '../../lib/email.js'
import type { Logger } from '../../lib/logger.js'
import type { ApiErrorCode, Friendship } from '../../types.js'
import type { FriendshipRecord, FriendshipsRepository } from './friendships.repository.js'

type RequestRefusal = 'user_not_found' | 'self' | 'already_friends' | 'request_exists'

const ERROR_BY_REFUSAL: Record<RequestRefusal, ApiErrorCode> = {
  user_not_found: 'USER_NOT_FOUND',
  self: 'SELF_FRIEND',
  already_friends: 'ALREADY_FRIENDS',
  request_exists: 'FRIEND_REQUEST_EXISTS',
}

// email comes straight from the JSON body: validated here, at the boundary.
export type FriendRequest = { email: unknown; requesterId: string }

export type FriendshipsServiceDeps = {
  repository: FriendshipsRepository
  logger: Logger
}

export type FriendshipsService = {
  requestFriendship: (request: FriendRequest) => Promise<Friendship>
}

// user is the other side: whoever reads the friendship already knows who he is.
function toFriendship(record: FriendshipRecord, userId: string): Friendship {
  return {
    id: record.id,
    user: record.requester.id === userId ? record.addressee : record.requester,
    createdAt: record.createdAt.toISOString(),
  }
}

// Ids and a reason only: never the email the sender typed.
function refusal(logger: Logger, reason: RequestRefusal, requesterId: string): AppError {
  logger.warn('friendship.request_failed', { reason, userId: requesterId })
  return new AppError(ERROR_BY_REFUSAL[reason])
}

async function findAddresseeId(
  { repository, logger }: FriendshipsServiceDeps,
  { email, requesterId }: FriendRequest,
): Promise<string> {
  const addresseeId = await repository.findUserIdByEmail(parseEmail(email))
  // Only logged-in members, all invited by the admin, can learn that an address has no account.
  if (addresseeId === null) {
    throw refusal(logger, 'user_not_found', requesterId)
  }
  if (addresseeId === requesterId) {
    throw refusal(logger, 'self', requesterId)
  }
  return addresseeId
}

async function rejectExistingPair(
  { repository, logger }: FriendshipsServiceDeps,
  requesterId: string,
  addresseeId: string,
): Promise<void> {
  const existing = await repository.findBetween(requesterId, addresseeId)
  if (existing?.status === 'accepted') {
    throw refusal(logger, 'already_friends', requesterId)
  }
  // Either direction: a reverse request already waits on the sender's page, with « Accepter ».
  if (existing?.status === 'pending') {
    throw refusal(logger, 'request_exists', requesterId)
  }
}

async function requestFriendship(
  deps: FriendshipsServiceDeps,
  request: FriendRequest,
): Promise<Friendship> {
  const { requesterId } = request
  const addresseeId = await findAddresseeId(deps, request)
  await rejectExistingPair(deps, requesterId, addresseeId)
  // The unique pairKey is the real guarantee: a concurrent request, in either order, ends here.
  const record = await deps.repository.create({ requesterId, addresseeId })
  if (!record) {
    throw refusal(deps.logger, 'request_exists', requesterId)
  }
  deps.logger.info('friendship.requested', { friendshipId: record.id, userId: requesterId })
  return toFriendship(record, requesterId)
}

export function createFriendshipsService(deps: FriendshipsServiceDeps): FriendshipsService {
  return {
    requestFriendship: (request) => requestFriendship(deps, request),
  }
}

import type { Request, Response } from 'express'
import { currentUser } from '../auth/auth.guard.js'
import type { FriendshipsService } from './friendships.service.js'

export type FriendshipsController = {
  list: (req: Request, res: Response) => Promise<void>
  request: (req: Request, res: Response) => Promise<void>
  remove: (req: Request<{ id: string }>, res: Response) => Promise<void>
}

export function createFriendshipsController(service: FriendshipsService): FriendshipsController {
  return {
    async list(req, res) {
      const list = await service.listFriendships(currentUser(req).id)
      res.json(list)
    },
    async request(req, res) {
      const friendship = await service.requestFriendship({
        email: req.body?.email,
        requesterId: currentUser(req).id,
      })
      res.status(201).json({ friendship })
    },
    async remove(req, res) {
      await service.deleteFriendship({ id: req.params.id, userId: currentUser(req).id })
      res.status(204).end()
    },
  }
}

import { errorMessage, UNAUTHORIZED_STATUS } from '@/api.ts'
import { type FriendshipList, isFriendshipList } from '@/friendship.ts'

export type FriendshipListResult =
  | { isSuccess: true; friendships: FriendshipList }
  | { isSuccess: false; message: string }

// The page reloads the whole list after every action: a success carries nothing more.
export type FriendshipActionResult = { isSuccess: true } | { isSuccess: false; message: string }

// A 401 calls onUnauthenticated: the session ended on the server, the provider drops the user.
export async function fetchFriendships(
  onUnauthenticated: () => void,
): Promise<FriendshipListResult> {
  try {
    const res = await fetch('/api/friendships')
    if (res.status === UNAUTHORIZED_STATUS) onUnauthenticated()
    // Not JSON when the dev proxy answers for a stopped API.
    const data: unknown = await res.json().catch(() => null)
    if (res.ok && isFriendshipList(data)) return { isSuccess: true, friendships: data }
    return { isSuccess: false, message: errorMessage(data) }
  } catch {
    // fetch rejects only on network failure.
    return { isSuccess: false, message: errorMessage(null) }
  }
}

async function sendAction(
  url: string,
  init: RequestInit,
  onUnauthenticated: () => void,
): Promise<FriendshipActionResult> {
  try {
    const res = await fetch(url, init)
    if (res.status === UNAUTHORIZED_STATUS) onUnauthenticated()
    if (res.ok) return { isSuccess: true }
    const data: unknown = await res.json().catch(() => null)
    return { isSuccess: false, message: errorMessage(data) }
  } catch {
    return { isSuccess: false, message: errorMessage(null) }
  }
}

export function postFriendRequest(
  email: string,
  onUnauthenticated: () => void,
): Promise<FriendshipActionResult> {
  const init: RequestInit = {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email }),
  }
  return sendAction('/api/friendships', init, onUnauthenticated)
}

import { useCallback, useEffect, useRef, useState } from 'react'
import { useAuth } from '@/auth/AuthContext.ts'
import type { FriendshipList } from '@/friendship.ts'
import { fetchFriendships } from '@/friendshipsApi.ts'

export type FriendshipsState = {
  friendships: FriendshipList
  isLoading: boolean
  error: string
  reload: () => Promise<void>
}

const NO_FRIENDSHIPS: FriendshipList = { friends: [], incoming: [], outgoing: [] }

export function useFriendships(): FriendshipsState {
  const { dropUser } = useAuth()
  const [friendships, setFriendships] = useState<FriendshipList>(NO_FRIENDSHIPS)
  const [isLoading, setIsLoading] = useState(true)
  const [error, setError] = useState('')
  // Two quick actions start two loads: only the latest one may set the lists (step 2 lesson).
  const latestLoadId = useRef(0)

  const reload = useCallback(async (): Promise<void> => {
    latestLoadId.current += 1
    const loadId = latestLoadId.current
    const result = await fetchFriendships(dropUser)
    if (loadId !== latestLoadId.current) return
    setIsLoading(false)
    setError(result.isSuccess ? '' : result.message)
    if (result.isSuccess) setFriendships(result.friendships)
  }, [dropUser])

  useEffect(() => {
    reload()
  }, [reload])

  return { friendships, isLoading, error, reload }
}

import { useState } from 'react'
import { useAuth } from '@/auth/AuthContext.ts'
import {
  deleteFriendship,
  type FriendshipActionResult,
  postAcceptFriendship,
} from '@/friendshipsApi.ts'

export type FriendshipActions = {
  // Ids with a call in flight: their buttons wait, the other rows stay usable.
  busyIds: string[]
  error: string
  accept: (id: string) => Promise<void>
  remove: (id: string) => Promise<void>
}

type SendAction = (id: string, onUnauthenticated: () => void) => Promise<FriendshipActionResult>

export function useFriendshipActions(reload: () => Promise<void>): FriendshipActions {
  const { dropUser } = useAuth()
  const [busyIds, setBusyIds] = useState<string[]>([])
  const [error, setError] = useState('')

  async function run(id: string, send: SendAction): Promise<void> {
    setError('')
    setBusyIds((ids) => [...ids, id])
    const result = await send(id, dropUser)
    setBusyIds((ids) => ids.filter((busyId) => busyId !== id))
    if (!result.isSuccess) setError(result.message)
    // Reload on failure too: a 404 means the other side acted first, the lists are stale.
    await reload()
  }

  return {
    busyIds,
    error,
    accept: (id) => run(id, postAcceptFriendship),
    remove: (id) => run(id, deleteFriendship),
  }
}
