> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Working agreement (mandatory):** keep this plan in sync with reality. Check a step only when it is finished and its gate is green. Append every divergence to the Drift Log. **Goal:** A member sends a photo to an accepted friend. From the library (a photo's actions menu), the friend gets a new `Photo` pointing to the same Cloudinary image: nothing is uploaded again. From the device (the upload form), the file is converted and uploaded as usual, but lands in the friend's library only. The friend's frame shows it at once with « Envoyée par X ». Each library deletes on its own: a received photo stays when the sender deletes the original or removes the friendship.

**Architecture:** `Photo` gains an optional `senderId` (named relations `PhotoOwner` / `PhotoSender` on `User`); the repository includes the sender's `displayName` only, and the DTO gains `senderName: string | null`. The photos service receives `areFriends` from the friendships service (built once in `app.ts`) and gains `shareFromLibrary` (copy of the original's `url`, `cloudinaryId`, `sha256` with the friend as owner) and `shareFromDevice` (the upload steps, factored as `storeNewPhoto`, with the friend as owner). Two routes: `POST /api/photos/:id/share` (JSON `{ friendId }`, its own `express.json()`) and `POST /api/photos/share` (multipart `photo` + `friendId`); both answer `201` with the friend's photo and emit `photo:new` to the friend's room only. Every bad `friendId` answers `403 NOT_FRIENDS`. The client gets the shadcn `Dialog`, `SharePhotoDialog` (accepted friends, one « Envoyer » per friend), an « Envoyer à un ami » entry in `PhotoActionsMenu` and in the upload form, and « Envoyée par X » on the caption, the list row and the labels. Four PRs in a `gh stack`.

**Tech Stack:** Node 24, TypeScript 7 (`nodenext`, `strict`, `verbatimModuleSyntax`), Express 5, Multer 2, Socket.IO 4, Prisma 6.19 (MongoDB), Vitest 5, Biome 2, Bruno CLI 4.2.0, React 19 + react-router 7 + shadcn/ui (radix-nova style, on the installed `radix-ui` package) + lucide-react (`SendIcon`), `gh stack`.

```ts
import { describe, expect, it, vi } from 'vitest'
import type { Logger } from '../../lib/logger.js'
import type { PhotoRecord, PhotosRepository } from './photos.repository.js'
import { createPhotosService, type PhotosServiceDeps } from './photos.service.js'

const OWNER_ID = '6b044f9d56a44767c7079b01'
const IMAGE = { url: 'https://res.cloudinary.com/demo/photo.jpg', cloudinaryId: 'familyframe/abc' }
const RECORD: PhotoRecord = {
  id: '6b044f9d56a44767c7079a01',
  ownerId: OWNER_ID,
  ...IMAGE,
  sha256: 'sha-of-the-original-bytes',
  createdAt: new Date('2026-11-23T10:00:00Z'),
}

// Every dependency succeeds by default; each test breaks only the one it is about.
// Deleting never converts or uploads: those fakes stay unused.
function makeDeps() {
  return {
    repository: {
      findPage: vi.fn<PhotosRepository['findPage']>(),
      findByOwnerAndSha256: vi.fn<PhotosRepository['findByOwnerAndSha256']>(),
      create: vi.fn<PhotosRepository['create']>(),
      deleteOwned: vi.fn<PhotosRepository['deleteOwned']>().mockResolvedValue(RECORD),
      countByCloudinaryId: vi.fn<PhotosRepository['countByCloudinaryId']>().mockResolvedValue(0),
    },
    toFrameJpeg: vi.fn<PhotosServiceDeps['toFrameJpeg']>(),
    uploadImage: vi.fn<PhotosServiceDeps['uploadImage']>(),
    deleteImage: vi.fn<PhotosServiceDeps['deleteImage']>().mockResolvedValue(undefined),
    logger: {
      info: vi.fn<Logger['info']>(),
      warn: vi.fn<Logger['warn']>(),
      error: vi.fn<Logger['error']>(),
    },
  }
}

describe('deletePhoto', () => {
  it("deletes the owner's record, then its image once no photo uses it, and returns the id", async () => {
    const deps = makeDeps()

    const deletedId = await createPhotosService(deps).deletePhoto({
      ownerId: OWNER_ID,
      id: RECORD.id,
    })

    expect(deletedId).toBe(RECORD.id)
    expect(deps.repository.deleteOwned).toHaveBeenCalledWith(RECORD.id, OWNER_ID)
    expect(deps.repository.countByCloudinaryId).toHaveBeenCalledWith(IMAGE.cloudinaryId)
    // Counted after the deletion: the deleted photo must not count itself.
    expect(deps.repository.deleteOwned.mock.invocationCallOrder[0]).toBeLessThan(
      deps.repository.countByCloudinaryId.mock.invocationCallOrder[0],
    )
    expect(deps.deleteImage).toHaveBeenCalledWith(IMAGE.cloudinaryId)
  })

  it('keeps the image while another photo still uses it', async () => {
    const deps = makeDeps()
    deps.repository.countByCloudinaryId.mockResolvedValue(1)

    await expect(
      createPhotosService(deps).deletePhoto({ ownerId: OWNER_ID, id: RECORD.id }),
    ).resolves.toBe(RECORD.id)
    expect(deps.deleteImage).not.toHaveBeenCalled()
    expect(deps.logger.info).toHaveBeenCalledWith('photo.deleted', {
      photoId: RECORD.id,
      cloudinaryId: RECORD.cloudinaryId,
      userId: OWNER_ID,
      isImageDestroyed: false,
    })
    expect(deps.logger.error).not.toHaveBeenCalled()
  })

  it('keeps the deletion and the image when the reference count fails', async () => {
    const deps = makeDeps()
    deps.repository.countByCloudinaryId.mockRejectedValue(new Error('database down'))

    await expect(
      createPhotosService(deps).deletePhoto({ ownerId: OWNER_ID, id: RECORD.id }),
    ).resolves.toBe(RECORD.id)
    expect(deps.deleteImage).not.toHaveBeenCalled()
    expect(deps.logger.error).toHaveBeenCalledWith('photo.orphaned', {
      cloudinaryId: IMAGE.cloudinaryId,
      error: expect.any(Error),
    })
```

```ts
export const DEFAULT_PAGE_SIZE = 30
export const MAX_PAGE_SIZE = 100
const DIGITS_PATTERN = /^\d+$/
const ALLOWED_TYPES: string[] = ['image/jpeg', 'image/png', 'image/webp']
const OBJECT_ID_PATTERN = /^[0-9a-f]{24}$/i

export type UploadedFile = { buffer: Buffer; mimetype: string }

export function parseFile(file: UploadedFile | undefined): UploadedFile {
  if (!file) {
    throw new AppError('PHOTO_MISSING')
  }
  // Cheap first filter: the browser's mimetype can lie, sharp checks for real.
  if (!ALLOWED_TYPES.includes(file.mimetype)) {
    throw new AppError('FORMAT_UNSUPPORTED')
  }
  return file
}

// Prisma fails on an id that is not an ObjectId: that would be a 500, not a 404.
export function parsePhotoId(id: string): string {
  if (!OBJECT_ID_PATTERN.test(id)) {
    throw new AppError('PHOTO_NOT_FOUND')
  }
  return id
}

// Express 5 gives a string, or an array for a repeated key: only one string of digits passes.
export function parseLimit(limit: unknown): number {
  if (limit === undefined) {
    return DEFAULT_PAGE_SIZE
  }
  if (typeof limit !== 'string' || !DIGITS_PATTERN.test(limit)) {
    throw new AppError('QUERY_INVALID')
  }
  const pageSize = Number(limit)
  if (pageSize < 1 || pageSize > MAX_PAGE_SIZE) {
    throw new AppError('QUERY_INVALID')
  }
  return pageSize
}

export function parseCursor(cursor: unknown): string | undefined {
  if (cursor === undefined) {
    return undefined
  }
  if (typeof cursor !== 'string' || !OBJECT_ID_PATTERN.test(cursor)) {
    throw new AppError('QUERY_INVALID')
  }
  return cursor
}
```

```ts
export type PhotosController = {
  list: (req: Request, res: Response) => Promise<void>
  create: (req: Request, res: Response) => Promise<void>
  remove: (req: Request<{ id: string }>, res: Response) => Promise<void>
}

type ControllerDeps = { service: PhotosService; io: SocketServer }

// cloudinaryId, ownerId and senderId stay on the server: the client gets ready-made URLs
// and the sender's name.
function toPhotoDto(record: PhotoRecord): Photo {
  return {
    id: record.id,
    ...deliveryUrls(record.cloudinaryId),
    createdAt: record.createdAt.toISOString(),
    senderName: record.sender?.displayName ?? null,
  }
}

// Only the owner's frames and pages: other accounts never see the photo.
function announceNewPhoto({ io }: ControllerDeps, res: Response, record: PhotoRecord): void {
  const photo = toPhotoDto(record)
  io.to(userRoom(record.ownerId)).emit('photo:new', photo)
  res.status(201).json(photo)
}

async function list({ service }: ControllerDeps, req: Request, res: Response): Promise<void> {
  const { records, nextCursor } = await service.listPage({
    ownerId: currentUser(req).id,
    cursor: req.query.cursor,
    limit: req.query.limit,
  })
  const page: PhotoPage = { photos: records.map(toPhotoDto), nextCursor }
  res.json(page)
}

async function create(deps: ControllerDeps, req: Request, res: Response): Promise<void> {
  const record = await deps.service.createPhoto({ ownerId: currentUser(req).id, file: req.file })
  announceNewPhoto(deps, res, record)
}

async function remove(
  { service, io }: ControllerDeps,
  req: Request<{ id: string }>,
  res: Response,
): Promise<void> {
  const ownerId = currentUser(req).id
  const deletedId = await service.deletePhoto({ ownerId, id: req.params.id })
  io.to(userRoom(ownerId)).emit('photo:deleted', deletedId)
  res.status(204).end()
}

export function createPhotosController(service: PhotosService, io: SocketServer): PhotosController {
  const deps: ControllerDeps = { service, io }
  return {
    list: (req, res) => list(deps, req, res),
    create: (req, res) => create(deps, req, res),
    remove: (req, res) => remove(deps, req, res),
```

```ts
import { useState } from 'react'
import { useAuth } from '@/auth/AuthContext.ts'
import type { Photo } from '@/photo.ts'
import { postDeviceShare, postLibraryShare, type ShareResult } from '@/photosApi.ts'

export type ShareTarget = { kind: 'library'; photo: Photo } | { kind: 'device'; file: File }

// Keyed by the friend's user id: one dialog may send to several friends.
export type FriendShares = {
  sendingIds: string[]
  sentIds: string[]
  error: string
  send: (friendId: string) => Promise<void>
}

function postShare(target: ShareTarget, friendId: string): Promise<ShareResult> {
  return target.kind === 'library'
    ? postLibraryShare(target.photo.id, friendId)
    : postDeviceShare(target.file, friendId)
}

export function useFriendShares(target: ShareTarget, onShared?: () => void): FriendShares {
  const { dropUser } = useAuth()
  const [sendingIds, setSendingIds] = useState<string[]>([])
  const [sentIds, setSentIds] = useState<string[]>([])
  const [error, setError] = useState('')

  async function send(friendId: string): Promise<void> {
    setSendingIds((ids) => [...ids, friendId])
    setError('')
    const result = await postShare(target, friendId)
    setSendingIds((ids) => ids.filter((id) => id !== friendId))
    // The session ended on the server: RequireAuth sends the user to /login.
    if (result.status === 'unauthenticated') {
      dropUser()
      return
    }
    if (result.status === 'failed') {
      setError(result.message)
      return
    }
    setSentIds((ids) => [...ids, friendId])
    onShared?.()
  }

  return { sendingIds, sentIds, error, send }
}
```
